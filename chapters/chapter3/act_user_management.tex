\begin{tikzpicture}[node distance=1.2cm, auto]

\tikzset{
    start/.style={circle, fill=blue!70!black, minimum size=12pt},
    action/.style={rectangle, draw, fill=teal!70!black, text=white, rounded corners=10pt, inner sep=6pt, text width=3cm, align=center, font=\footnotesize\bfseries},
    decision/.style={diamond, draw, fill=cyan!60!black, text=white, aspect=2, inner sep=2pt, text width=2.2cm, align=center, font=\scriptsize\bfseries},
    fork/.style={rectangle, fill=blue!80!black, minimum width=9cm, minimum height=0.15cm},
    stop/.style={circle, draw=black, double, fill=gray!70, minimum size=10pt},
    arrow/.style={-Stealth, thick, draw=gray!70},
    labelNode/.style={font=\scriptsize\bfseries, fill=teal!50, inner sep=1pt}
}

\node [start] (start) {};
\node [action, below=of start] (signin) {Sign in};
\node [decision, below=of signin] (checkSign) {Sign in Successful?};
\node [decision, below=1.5cm of checkSign] (checkRole) {User Role?};

\node [fork, below=1.5cm of checkRole] (mainFork) {};

\node [action, left=2.8cm of mainFork, anchor=north] (viewApp) {View Applicant List};
\node [action, below=0.2cm of mainFork.center] (viewUser) {View User Information};
\node [action, right=2.8cm of mainFork, anchor=north] (handleEdit) {Handle Edit Request};

\node [fork, below=0.8cm of viewUser, minimum width=6cm] (subFork) {};

\node [action, left=0.8cm of subFork, anchor=north] (delUser) {Delete User};
\node [decision, below=of delUser] (confirmDel) {Delete User?};
\node [action, below=of confirmDel] (succDel) {Successfully Delete User};

\node [action, right=0.8cm of subFork, anchor=north] (updUser) {Update User};
\node [decision, below=of updUser] (confirmUpd) {Update User?};
\node [action, below=of confirmUpd] (succUpd) {Successfully Update User};

\node [stop, below=8.5cm of viewApp] (stop1) {};
\node [stop, below=1cm of succDel] (stop2) {};
\node [stop, below=1cm of succUpd] (stop3) {};
\node [stop, below=8.5cm of handleEdit] (stop4) {};

\draw [arrow] (start) -- (signin);
\draw [arrow] (signin) -- (checkSign);

\draw [arrow] (checkSign.west) -- ++(-0.8,0) |- node[labelNode, pos=0.2] {No} (signin.west);
\draw [arrow] (checkSign) -- node[labelNode] {Yes} (checkRole);
\draw [arrow] (checkRole.east) -- ++(1.2,0) |- node[labelNode, pos=0.2] {No} (signin.east);
\draw [arrow] (checkRole) -- node[labelNode] {Yes} (mainFork);

\draw [arrow] (mainFork.west) ++(1.7,0) -- (viewApp);
\draw [arrow] (mainFork.center) -- (viewUser);
\draw [arrow] (mainFork.east) ++(-1.7,0) -- (handleEdit);

\draw [arrow] (viewUser) -- (subFork);
\draw [arrow] (subFork.west) ++(2.2,0) -- (delUser);
\draw [arrow] (subFork.east) ++(-2.2,0) -- (updUser);

\draw [arrow] (delUser) -- (confirmDel);
\draw [arrow] (confirmDel) -- node[labelNode] {Yes} (succDel);
\draw [arrow] (confirmDel.west) -- ++(-0.5,0) |- node[labelNode, pos=0.2] {No} (delUser.west);

\draw [arrow] (updUser) -- (confirmUpd);
\draw [arrow] (confirmUpd) -- node[labelNode] {Yes} (succUpd);
\draw [arrow] (confirmUpd.east) -- ++(0.5,0) |- node[labelNode, pos=0.2] {No} (updUser.east);

\draw [arrow] (viewApp) -- (stop1);
\draw [arrow] (handleEdit) -- (stop4);
\draw [arrow] (succDel) -- (stop2);
\draw [arrow] (succUpd) -- (stop3);

\end{tikzpicture}
